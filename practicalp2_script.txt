-- 1. Eliminar la base de datos si existe
DROP DATABASE IF EXISTS practicalp2;

-- 2. Crear la base de datos
CREATE DATABASE practicalp2;

-- 3. Seleccionar la base de datos
USE practicalp2;

-- 4. Crear tabla categoria
CREATE TABLE categoria (
    id_categoria INT NOT NULL PRIMARY KEY,
    descripcion VARCHAR(100),
    nombre VARCHAR(50)
);

-- 5. Crear tabla plataforma
CREATE TABLE plataforma (
    id_plataforma INT NOT NULL PRIMARY KEY,
    fabricante VARCHAR(50),
    nombre VARCHAR(50)
);

-- 6. Crear tabla proveedor
CREATE TABLE proveedor (
    id_proveedor INT NOT NULL PRIMARY KEY,
    email VARCHAR(100),
    nombre VARCHAR(100),
    telefono VARCHAR(15)
);

-- 7. Crear tabla videojuego
CREATE TABLE videojuego (
    id_juego VARCHAR(10) NOT NULL PRIMARY KEY,
    id_categoria INT NOT NULL,
    id_plataforma INT NOT NULL,
    id_proveedor INT NOT NULL,
    precio DOUBLE NOT NULL,
    stock INT NOT NULL,
    titulo VARCHAR(100),
    
    -- Claves foráneas
    FOREIGN KEY (id_categoria) REFERENCES categoria(id_categoria),
    FOREIGN KEY (id_plataforma) REFERENCES plataforma(id_plataforma),
    FOREIGN KEY (id_proveedor) REFERENCES proveedor(id_proveedor)
);
